project(shift_engine VERSION 0.0.1 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(SHIFT_BUILD_CARVIEW "Build the carview app (SDL3 + Vulkan)" ON)
option(SHIFT_BUILD_LAUNCHER "Build the unified cross-platform GUI launcher" ON)
option(SHIFT_ENABLE_RAY_TRACING "Compile Vulkan hardware ray tracing & path tracing backend" ON)
option(SHIFT_STATIC_RUNTIME "Link libstdc++/libgcc statically on Linux" ON)

if(MSVC)
  add_compile_options(/utf-8 /permissive- /Zc:__cplusplus $<$<COMPILE_LANGUAGE:CXX>:/EHsc>)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND SHIFT_STATIC_RUNTIME)
  add_link_options(-static-libstdc++ -static-libgcc)
endif()
if(ANDROID)
  # Universal ARMv8.0-A compatibility for lowest requirements, tuned for Cortex-X3
  # (Snapdragon 8 Gen 2 / Odin 2) with ThinLTO + section GC + safe ICF in Release.
  add_compile_options($<$<CONFIG:Release>:-flto=thin> $<$<CONFIG:Release>:-fno-semantic-interposition>
                      $<$<CONFIG:Release>:-ffunction-sections> $<$<CONFIG:Release>:-fdata-sections>)
  if(CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
    add_compile_options($<$<CONFIG:Release>:-mtune=cortex-x3>)
  endif()
  add_link_options($<$<CONFIG:Release>:-flto=thin> $<$<CONFIG:Release>:-Wl,--gc-sections>
                   $<$<CONFIG:Release>:-Wl,--icf=safe>)
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(Dependencies)

# Refresh SHIFT_GIT_COMMIT automatically whenever HEAD or its active branch ref moves.
set(_shift_git_dir "${CMAKE_CURRENT_SOURCE_DIR}/.git")
if(EXISTS "${_shift_git_dir}" AND NOT IS_DIRECTORY "${_shift_git_dir}")
  file(READ "${_shift_git_dir}" _shift_gitdir_line)
  string(STRIP "${_shift_gitdir_line}" _shift_gitdir_line)
  if(_shift_gitdir_line MATCHES "^gitdir:[ \t]*(.+)$")
    set(_shift_git_dir "${CMAKE_MATCH_1}")
    if(NOT IS_ABSOLUTE "${_shift_git_dir}")
      get_filename_component(_shift_git_dir "${CMAKE_CURRENT_SOURCE_DIR}/${_shift_git_dir}" ABSOLUTE)
    endif()
  endif()
endif()
if(EXISTS "${_shift_git_dir}/HEAD")
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_shift_git_dir}/HEAD")
  file(READ "${_shift_git_dir}/HEAD" _shift_git_head)
  string(STRIP "${_shift_git_head}" _shift_git_head)
  if(_shift_git_head MATCHES "^ref:[ \t]*(.+)$")
    set(_shift_git_ref "${_shift_git_dir}/${CMAKE_MATCH_1}")
    if(EXISTS "${_shift_git_ref}")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_shift_git_ref}")
    elseif(EXISTS "${_shift_git_dir}/packed-refs")
      set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_shift_git_dir}/packed-refs")
    endif()
  endif()
endif()
